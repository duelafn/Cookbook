
\begin{recipe}{Spaghetti}%\tag{}
  \rating    4
  \nutrition
  \health
  \workload{medium}
  \time{60}
  \yield{}
  \servings{4}
  \source{Dean}
  \photo{spaghetti}
  \maketitle

  \begin{ingredients2}
    8 \oz & mushrooms\\
    2 cloves & garlic\\
    1 & zucchini\\
    1 \T & \reciperef{Italian Sausage Seasoning}\\
    1 \lb & ground beef or pork\\
    12 \oz & noodles\\
    24 \oz jar & spaghetti sauce\\
    & shredded carrot\\
  \end{ingredients2}

  Start water for noodles, cook noodles while cooking other things. Cook mushrooms.
  Cook zucchini with garlic. Cook meat with seasoning and shredded carrots. Stir
  everything together when done.
\end{recipe}
