
\begin{recipe}{Popcorn}
  \rating    5
  \nutrition 3
  \health    2
  \workload{light}
  \time{15}
  \yield{10~cups}
  \servings{2}
  \source{}
  % \photo{}
  \maketitle

  \begin{ingredients2}
    \half \cup & poppcorn\\
    3 \T & butter (melted)\\
    1\half pinch & salt
  \end{ingredients2}

  If using an air popper, start pouring butter while still popping for even coverage.
\end{recipe}

\begin{recipe}{Kettle Corn}
  \rating    4
  \nutrition 3
  \health    2
  \workload{light}
  \time{15}
  \yield{10~cups}
  \servings{2}
  \source{}
  % \photo{}
  \maketitle

  \begin{ingredients2}
    \half \cup & poppcorn\\
    \quarter \c & peanut or canola oil\\
    \quarter \c & sugar\\
    pinch & salt
  \end{ingredients2}

  Thin and crispy candy coating, individual kernels.

  Heat oil over medium-high heat in a heavy pot large enough to contain popped corn
  (5 qt at least). Add all ingredients, give a quick stir and cover with lid.

  After first kernel pops, shake the pan to mix (do not uncover) for 3 seconds. Shake
  again every 10 seconds until popping begins to slow (2--3 seconds between pops).

  Immediately remove from heat and dump onto a tray to cool.
\end{recipe}

\begin{recipe}{Popcorn Balls}
  \rating    4
  \nutrition 3
  \health    2
  \workload{medium}
  \time{30}
  \yield{10~cups}
  \servings{2}
  \source{}
  % \photo{}
  \maketitle

  \begin{ingredients2}
    1 \cup & poppcorn (kernels)\\
    \fourth \c & water\\
    \threefourth \cup & sugar\\
    \threefourth \cup & corn syrup\\
    3 \T & butter\\
    1 \T & vanilla\\
    pinch & salt
  \end{ingredients2}

  Chewy candy, popcorn clusters.

  Pop popcorn in air popper, spread out in large sheet pans (recommend wax paper in
  the bottom of the pans). Heat sugars and water in small sauce pan to 240\degF. Once
  temperature reached, allow to cool a bit and stir in butter, then stir in vanilla
  and salt. Drizzle over popcorn, optionally form into balls.
\end{recipe}
