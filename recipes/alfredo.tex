% Fake: roux
% Real: Cream

\begin{recipe}{Fake Alfredo}%\tag{}
  \rating    0
  \nutrition 0
  \health    2
  \workload{medium}
  \time{}
  \yield{}
  \servings{}
  \source{}
  \photo{alfredo}
  \maketitle

  \begin{ingredients2}
    8 \oz & mushrooms\\
    2 cloves & garlic\\
    & blackened chicken or salmon\\
    2 \T & butter\\
    2 \T & flour\\
    1\half \c & milk\\
    5 \oz & parmesan\\
    & black pepper\\
    \threefourth \lb & pasta
  \end{ingredients2}

  Cook mushrooms, then chicken and garlic.

  Deglaze with butter and add flour and cook roux. Add milk slowly until smooth and
  consistency reached. Add parmesan and pepper. Toss with noodles.
\end{recipe}
